-- Q1_TopicGopYNhieuNhat --
PREFIX ntu: <http://ntu.edu.vn/feedback_system#>
SELECT ?topic (COUNT(?fb) AS ?soLuong)
WHERE {
    ?fb a ntu:Feedback .
    ?fb ntu:aboutTopic ?topicNode .
    ?topicNode rdfs:label ?topic .
    FILTER (?topic != "Khac")
}
GROUP BY ?topic
ORDER BY DESC(?soLuong)

-- Q2a_TieuCucTheoGV_Count --
PREFIX ntu: <http://ntu.edu.vn/feedback_system#>
SELECT ?maGV (COUNT(?fb) AS ?soLuongTieuCuc)
WHERE {
    ?fb a ntu:Feedback .
    ?fb ntu:aboutLecturer ?gvNode .
    ?gvNode ntu:hasLecturerCode ?maGV .
    ?fb ntu:hasSentiment ntu:Sent_Negative .
}
GROUP BY ?maGV
ORDER BY DESC(?soLuongTieuCuc)

-- Q8_PhanHoiThatVong --
PREFIX ntu: <http://ntu.edu.vn/feedback_system#>
SELECT ?fb ?giangVien ?hocPhan ?noiDung
WHERE {
    ?fb a ntu:Feedback .
    ?fb ntu:hasEmotion ntu:Emo_ThatVong .
    ?fb ntu:hasContent ?noiDung .
    OPTIONAL {
        ?fb ntu:aboutLecturer ?gvNode .
        ?gvNode ntu:hasLecturerCode ?giangVien .
    }
    OPTIONAL {
        ?fb ntu:aboutCourse ?courseNode .
        ?courseNode ntu:hasCourseCode ?hocPhan .
    }
}
